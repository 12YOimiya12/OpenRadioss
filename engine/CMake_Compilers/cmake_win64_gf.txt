# Fortran <-> C symbol naming bridge
# ----------------------------------
# The C/C++ sources declare Fortran entry points using the Intel-on-Windows
# convention (UPPERCASE, no trailing underscore) - see the "#ifdef _WIN64"
# blocks in engine/share/includes/dyn_userlib.h, engine/source/user_interface/
# dyn_userlib_callback.c, common_source/tools/set_operation/set_tools.cpp and
# similar. gfortran instead emits the GNU convention (lowercase + trailing
# underscore), so the two spellings never meet and the link fails with ~80
# undefined references.
#
# Rather than edit those ~45 files, the linker is told to define the missing
# expected spelling as an alias of the symbol gfortran actually emitted:
#     ld --defsym,ARRET=arret_
# --defsym is a plain symbol table definition (no thunk, no extra call frame)
# and works on PE/COFF. Both directions occur: uppercase names referenced from
# C but emitted lowercase by gfortran, and the reverse for the user-library
# callbacks that Fortran calls back into C.
#
# The reverse-direction entries below are exactly the aliases the sources
# already declare under "#ifdef _WIN64" in engine/share/includes/dyn_userlib.h
# and common_source/tools/set_operation/set_tools.cpp.
set ( fortran_names

# Stack reserve
# -------------
# The sources keep large arrays as locals, e.g. IWORK(70000) (280 kB) in
# starter/source/system/sysfus.F, so the default 1-2 MB Windows stack is not
# enough and the process dies in ___chkstk_ms during a routine's prologue.
# This is not a MinGW quirk: the Intel Windows configuration
# (engine/CMake_Compilers/cmake_win64.txt) passes /STACK:1500000000 for the
# same reason, and the value below matches it.
set ( win_stack "-Wl,--stack,1500000000" )

# Linking flags
# fortran_names is a CMake list; joining it into a quoted string would leave
# the list separators (";") in place, which ld rejects. Expand it first.
string (REPLACE ";" " " fortran_names "${fortran_names}")
set (CMAKE_EXE_LINKER_FLAGS "${OPENMP} ${FSANITIZE} ${fortran_names} ${win_stack}" )

#Libraries

# NOTE: LINK is a CMake *list* of items, not one space separated string.
# A single quoted string makes the Ninja generator emit
#   LINK_LIBRARIES = "<all files in one quoted token>"
# which ninja then tries to stat as a single (absurdly long) filename.
if ( static_link STREQUAL "1" )
  set (LINK ${metis_lib} ${lapack_lib} ${zlib_lib} ${md5_lib} ${mpi_lib}
            ws2_32 -static-libgfortran -static-libstdc++ -static-libgcc)
else()
  set (LINK ${metis_lib} ${lapack_lib} ${zlib_lib} ${md5_lib} ${mpi_lib} ws2_32)
endif()
